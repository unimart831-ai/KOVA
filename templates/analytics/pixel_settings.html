{% endif %} {# ── Setup Instructions ── #}

Installation

Add this snippet to your website, just before the closing </head> tag. 

<!-- Kova Pixel -->
<script>
  window._kovaConfig = {
    token: '{{ pixel_token }}',
    endpoint: '{{ base_url }}/analytics/pixel/track/'
  };
</script>
<script src="{{ base_url }}{% url 'analytics:pixel_js' %}" defer></script>
